# The code should have functionality to create users. Users can be either customer or admin. All users will have a unique 
# identifier: username.

# Admin can create a Loan in the system for a customer.

# While creating a loan, admin_username, customer_username, principal amount, interest rate and time (loan tenure) need to be 
# taken as input.

# The interest for the loan is calculated by I = (P*N*R)/100 where P is the principal amount, N is the number of years and R 
# is the rate of interest. The total amount to repay will be A = P + I The amount should be paid back monthly in the form of 
# EMIs. Each EMI = A/(N*12)

# Users should be able to make EMI payments for their loan only.

# Users should be able to fetch loan info for their loans only. Fetching a loan should return the loan info along with all 
# the emi payments done and EMIs remaining[optional].

# Admin should be able to fetch all loans for all customers.

# All the functions should take username as one of the arguments, and all user level validation should happen against this
#  username.

class User:
    def __init__(self, username, role): # role = customer or admin
        self.username = username
        self.role = role

class Loan:
    def __init__(self, principal, rate, years):
        self.principal = principal
        self.rate = rate
        self.years = years

        self.interest = (principal * years * rate) / 100 # I = (P*N*R)/100
        self.total_amount = self.principal + self.interest # A = P + I
        self.total_emis = years * 12
        self.emi = self.total_amount / self.total_emis  # EMI = A/(N*12)

        self.paid_emis = 0

        def pay_emi(self):
            if self.paid_emis < self.total_emis:
                self.paid_emis += 1
                print("paid emi")
            else:
                print("loan paid")

        def get_balance(self):
            paid = self.paid_emis * self.emi
            remaining = self.total_amount - paid

            return remaining

class LoanCalculator:
    def __init__(self):
        self.users = {}
        self.loans = {}
    
    def create_user(self, username, role):
        if username not in self.users:
            self.users[username] = User(username, role)
            print("user created: ", username, role)
        else:
            print("user not created, already exist")
    
    def create_loan(self, username, principal, rate, years):
        if username in self.users and self.users[username].role == "customer":
            self.loans[username] = Loan(principal, rate, years)
            print("loan created: ",  self.loans[username].emi)
        else:
            print("role should be customer")


    def pay_emis(self, username):

        loan = self.loans.get(username)
        print("loan", loan)
        if loan:
            loan.pay_emi()
        else:
            print("loan not found")

    def check_balance(self, username):
        loan = self.loans.get(username)
        if loan:
            return loan.get_balance()
        return "balance not found"


loan = LoanCalculator()
loan.create_user("manish", "customer")
loan.create_loan("manish", 10, 10, 10) # username, principal, rate, years
loan.pay_emis("manish")
print("balance", loan.check_balance("manish"))
