# Simple Chess Game in Python (Enforces Legal Moves)
board = [
    ["r", "n", "b", "q", "k", "b", "n", "r"],
    ["p", "p", "p", "p", "p", "p", "p", "p"],
    [".", ".", ".", ".", ".", ".", ".", "."],
    [".", ".", ".", ".", ".", ".", ".", "."],
    [".", ".", ".", ".", ".", ".", ".", "."],
    [".", ".", ".", ".", ".", ".", ".", "."],
    ["P", "P", "P", "P", "P", "P", "P", "P"],
    ["R", "N", "B", "Q", "K", "B", "N", "R"]
]

def print_board():
    for i, row in enumerate(board, start=1):
        print(f"{8-i+1} {' '.join(row)}")
    print("  a b c d e f g h\n")

def get_position(pos):
    return 8 - int(pos[1]), ord(pos[0]) - ord('a')

def is_valid_move(start, end, piece):
    x1, y1 = start
    x2, y2 = end
    dx, dy = abs(x2 - x1), abs(y2 - y1)

    if piece.lower() == "p":  # Pawn
        direction = -1 if piece.isupper() else 1
        if y1 == y2 and board[x2][y2] == ".":  # Forward move
            return (dx == 1) or (dx == 2 and (x1 == 1 or x1 == 6))
        if dx == 1 and dy == 1 and board[x2][y2] != ".":  # Capture
            return True

    elif piece.lower() == "r":  # Rook
        if x1 == x2 or y1 == y2:  # Straight move
            return is_clear_path(start, end)
    
    elif piece.lower() == "b":  # Bishop
        if dx == dy:  # Diagonal move
            return is_clear_path(start, end)
    
    elif piece.lower() == "q":  # Queen
        if dx == dy or x1 == x2 or y1 == y2:  # Rook or Bishop move
            return is_clear_path(start, end)

    elif piece.lower() == "k":  # King
        return dx <= 1 and dy <= 1  # One square move
    
    elif piece.lower() == "n":  # Knight (L-shape)
        return (dx, dy) in [(2, 1), (1, 2)]
    
    return False

def is_clear_path(start, end):
    x1, y1 = start
    x2, y2 = end
    dx = (x2 - x1) // max(1, abs(x2 - x1))
    dy = (y2 - y1) // max(1, abs(y2 - y1))
    
    x, y = x1 + dx, y1 + dy
    while (x, y) != (x2, y2):
        if board[x][y] != ".":
            return False
        x += dx
        y += dy
    return True

def move_piece(start, end):
    x1, y1 = get_position(start)
    x2, y2 = get_position(end)
    
    piece = board[x1][y1]
    
    if piece == ".":
        print("No piece at the starting position!")
        return False

    if board[x2][y2].isupper() == piece.isupper():
        print("Cannot capture your own piece!")
        return False

    if not is_valid_move((x1, y1), (x2, y2), piece):
        print("Invalid move for this piece!")
        return False

    board[x2][y2] = piece
    board[x1][y1] = "."
    return True

print_board()

while True:
    move = input("Enter move (e.g., e2 e4) or 'q' to quit: ")
    if move.lower() == "q":
        break
    try:
        start, end = move.split()
        if move_piece(start, end):
            print_board()
    except Exception:
        print("Invalid input format. Try again.")
