SELECT 
c.first_name,
c.last_name,
o.item,
s.status
FROM Customers c
JOIN Shippings s
ON c.customer_id = s.customer
LEFT JOIN Orders o
ON c.customer_id = o.customer_id
WHERE s.status = 'Pending';

FROM Customers

FROM Customers

FROM Customers

FROM Customers;

FROM Customers

SELECT first_name,
CASE
    WHEN age >= 30 THEN 'Senior'
    ELSE 'Young'
END AS category
FROM Customers;

SELECT item, amount
FROM Orders
ORDER BY amount DESC;

SELECT SUM(amount) AS total_sales
FROM Orders;

SELECT customer_id, SUM(amount) AS total_spent
FROM Orders
GROUP BY customer_id;

SELECT customer_id, SUM(amount) AS total_spent
FROM Orders
GROUP BY customer_id
HAVING SUM(amount) > 500;

SELECT *
FROM Orders
WHERE amount > (
    SELECT AVG(amount)
    FROM Orders
);

SELECT
order_id,
item,
amount,
RANK() OVER (ORDER BY amount DESC) AS price_rank
FROM Orders;

SELECT
item,
amount,
CASE
WHEN amount >= 1000 THEN 'Expensive'
WHEN amount >= 400 THEN 'Medium'
ELSE 'Cheap'
END AS category
FROM Orders;


SELECT
c.first_name,
SUM(o.amount) AS total_spent
FROM Customers c
JOIN Orders o
ON c.customer_id = o.customer_id
GROUP BY c.first_name
HAVING SUM(o.amount) > 500
ORDER BY total_spent DESC;
